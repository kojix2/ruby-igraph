def self.check_error(result)
  return result if result == LibIGraph::IGRAPH_SUCCESS

  msg = LibIGraph.igraph_strerror(result) || "Unknown igraph error (code=#{result})"
  msg = last_c_error || msg
  clear_last_c_error

  case result
  when LibIGraph::IGRAPH_ENOMEM
    raise IGraph::NoMemoryError, msg
  when LibIGraph::IGRAPH_EINVAL
    raise IGraph::ArgumentError, msg
  when LibIGraph::IGRAPH_EINVVID, LibIGraph::IGRAPH_EINVEVECTOR, LibIGraph::IGRAPH_EINVMODE
    raise IGraph::ArgumentError, msg
  when LibIGraph::IGRAPH_EFILE, LibIGraph::IGRAPH_PARSEERROR
    raise IGraph::IOError, msg
  else
    raise IGraph::RuntimeError, msg
  end
end

def search_igraph(name = nil)
  name ||= "libigraph.#{FFI::Platform::LIBSUFFIX}"
  lib_path = if ENV["IGRAPHDIR"]
               File.expand_path(name, ENV["IGRAPHDIR"])
             else
               File.expand_path("../vendor/#{name}", __dir__)
             end
  return lib_path if File.exist?(lib_path)

  begin
    require "pkg-config"
    lib_dir = PKGConfig.variable("igraph", "libdir")
    lib_path = File.expand_path(name, lib_dir)
  rescue PKGConfig::NotFoundError
    warn "igraph.pc was not found in the pkg-config search path."
  end
  return lib_path if File.exist?(lib_path)

  warn "igraph shared library '#{name}' not found."
end

def search_igraph_windows
  ENV["IGRAPHDIR"] ||= [
    RubyInstaller::Runtime.msys2_installation.msys_path,
    RubyInstaller::Runtime.msys2_installation.mingwarch
  ].join(File::ALT_SEPARATOR)
  path = File.expand_path("bin/libigraph.dll", ENV["IGRAPHDIR"])
  RubyInstaller::Runtime.add_dll_directory(File.dirname(path))
  path
end

def self.setup_error_handler
  return if @error_handler_setup
  @error_handler_setup = true

  handler = FFI::Function.new(:void, %i[string string int int]) do |reason, _file, _line, _errno|
    @last_c_error = reason
  end
  LibIGraph.igraph_set_error_handler(handler)
  @error_handler_func = handler # Prevent GC
end
